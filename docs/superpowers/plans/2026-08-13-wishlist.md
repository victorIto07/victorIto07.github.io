# Wishlist Page Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build a mobile-first birthday wishlist page where gifts orbit around a focal point and friends can anonymously reserve one so nobody buys a duplicate.

**Architecture:** A static page on GitHub Pages, no build step. Pure geometry and identity logic lives in `lib.js` and is unit-tested under Node. The DOM app in `index.html` is split into four non-overlapping pieces — `orbit` (rotation and painting), `card` (detail display), `claims` (the only code that touches Firebase), and `app` (wiring plus the name gate). Isolating `claims` is what lets the page degrade to a working stateless showcase when Firebase is unreachable.

**Tech Stack:** Vanilla ES5-compatible JS, no framework, no bundler. Firebase Realtime Database via the compat CDN build. Node (any version with `node:test`) for unit tests only — it is never needed to serve the page.

**Spec:** `docs/superpowers/specs/2026-08-13-wishlist-design.md`

## Global Constraints

- Page lives at `wishlist/`, served as `https://victorito07.github.io/wishlist/`.
- No build step. Files are served exactly as committed.
- All user-facing copy is Brazilian Portuguese.
- `lib.js` assigns `globalThis.WL` and contains **no `export` statements**, so it loads as a browser `<script>` and as CommonJS under Node.
- `itens.js` assigns `window.ITENS`. Never `fetch()` it — script tags work from `file://`, `fetch` does not.
- No ES modules in browser code, no `import`/`export` outside `test.mjs`.
- Firebase path is `/claims/<itemId>`, value is a hashed name string. Nothing else is ever written.
- `crypto.subtle` is forbidden — it is unavailable over `file://`. Use cyrb53 from `lib.js`.
- Never store a raw or normalized name in Firebase, only the hash.
- Viewport height uses `100dvh`, never `100vh`.
- Respect `prefers-reduced-motion`: no idle drift, no bobbing, instant snapping.

```js
/* Pure helpers. No DOM, no network. Loads as a browser <script> and as
   CommonJS under Node — do not add `export` statements. */
(function (root) {
  'use strict';

  function normalizeName(raw) {
    if (typeof raw !== 'string') return '';
    return raw
      .toLowerCase()
      .normalize('NFD')
      .replace(/[\u0300-\u036f]/g, '')   // combining diacritics
      .replace(/[^a-z0-9 ]/g, '')
      .replace(/\s+/g, ' ')
      .trim();
  }

  /* cyrb53 — deterministic ~53-bit string hash.
     Deliberately not crypto.subtle: that needs a secure context and is
     unavailable over file://, which would make local previews produce
     different identities than the deployed page. Hash strength is
     irrelevant here anyway — the input space is a few dozen first names. */
  function hashName(str) {
    var s = typeof str === 'string' ? str : '';
    var h1 = 0xdeadbeef, h2 = 0x41c6ce57;
    for (var i = 0; i < s.length; i++) {
      var ch = s.charCodeAt(i);
      h1 = Math.imul(h1 ^ ch, 2654435761);
      h2 = Math.imul(h2 ^ ch, 1597334677);
    }
    h1 = Math.imul(h1 ^ (h1 >>> 16), 2246822507) ^ Math.imul(h2 ^ (h2 >>> 13), 3266489909);
    h2 = Math.imul(h2 ^ (h2 >>> 16), 2246822507) ^ Math.imul(h1 ^ (h1 >>> 13), 3266489909);
    var n = 4294967296 * (2097151 & h2) + (h1 >>> 0);
    return n.toString(16);
  }

  root.WL = { normalizeName: normalizeName, hashName: hashName };
})(typeof globalThis !== 'undefined' ? globalThis : this);
```

### Task 2: Pure logic core (`lib.js`) — orbit geometry

**Files:**
- Modify: `wishlist/lib.js`
- Modify: `wishlist/test.mjs`

**Interfaces:**
- Consumes: `globalThis.WL` from Task 1.
- Produces: `WL.selectedIndex(angle, count) -> number`, `WL.snapTarget(angle, count) -> number`, `WL.layout(angle, i, count, geom) -> {x, y, scale, opacity, z}`.

Geometry model: item `i` sits at angle `angle + i * (2π / count)`. It is in
front when that value is a multiple of 2π. Depth runs 0 (back) to 1 (front)
as `(cos(a) + 1) / 2`.

```js
  var TAU = Math.PI * 2;

  function selectedIndex(angle, count) {
    var step = TAU / count;
    var i = Math.round(-angle / step) % count;
    return (i + count) % count;
  }

  function snapTarget(angle, count) {
    var step = TAU / count;
    return Math.round(angle / step) * step;
  }

  function layout(angle, i, count, geom) {
    var a = angle + i * (TAU / count);
    var depth = (Math.cos(a) + 1) / 2;
    var scale = 0.48 + depth * 0.52;
    return {
      x: geom.cx + Math.sin(a) * geom.rx - geom.size / 2,
      y: geom.cy - Math.cos(a) * geom.ry - geom.size / 2 - depth * geom.lift,
      scale: scale,
      opacity: 0.26 + depth * 0.74,
      z: Math.round(depth * 100)
    };
  }
```

Then extend the export to:

```js
  root.WL = {
    normalizeName: normalizeName,
    hashName: hashName,
    selectedIndex: selectedIndex,
    snapTarget: snapTarget,
    layout: layout
  };
```

```html
<script>
(function () {
  'use strict';
  var TAU = Math.PI * 2;
  var items = Array.isArray(window.ITENS) ? window.ITENS : [];
  var orb = document.getElementById('orb');
  var reduce = window.matchMedia('(prefers-reduced-motion: reduce)').matches;

  var GEOM = { cx: 0, cy: 132, rx: 98, ry: 44, size: 82, lift: 24 };
  var els = [], ang = 0, vel = 0, target = 0, snapping = false;
  var drag = false, lastX = 0, moved = 0, sel = -1;

  function build() {
    orb.innerHTML = '';
    els = [];
    items.forEach(function (it, i) {
      var d = document.createElement('div');
      d.className = 'oi';
      d.style.width = GEOM.size + 'px';
      d.style.height = GEOM.size + 'px';
      d.innerHTML = '<div class="ph">🎁</div>';
      if (it.img) {
        var img = document.createElement('img');
        img.alt = '';
        img.addEventListener('error', function () { img.remove(); });
        img.src = it.img;
        d.appendChild(img);
      }
      d.addEventListener('click', function () {
        if (moved > 6) return;      // it was a drag, not a tap
        goTo(i);
      });
      orb.appendChild(d);
      els.push(d);
    });
  }

  function goTo(i) {
    var step = TAU / items.length;
    var k = Math.round(ang / step), want = -i, best = want, bd = Infinity;
    for (var t = -2; t <= 2; t++) {
      var c = want + t * items.length, dd = Math.abs(c - k);
      if (dd < bd) { bd = dd; best = c; }
    }
    target = best * step;
    if (reduce) { ang = target; snapping = false; } else { snapping = true; }
    vel = 0;
  }

  orb.addEventListener('pointerdown', function (e) {
    drag = true; snapping = false; lastX = e.clientX; vel = 0; moved = 0;
    orb.setPointerCapture(e.pointerId);
  });
  orb.addEventListener('pointermove', function (e) {
    if (!drag) return;
    var dx = e.clientX - lastX;
    moved += Math.abs(dx);
    vel = dx * 0.0042; ang += vel; lastX = e.clientX;
  });
  function release() {
    if (!drag) return;
    drag = false;
    if (reduce || Math.abs(vel) < 0.004) {
      target = WL.snapTarget(ang, items.length);
      if (reduce) { ang = target; snapping = false; } else { snapping = true; }
    }
  }
  orb.addEventListener('pointerup', release);
  orb.addEventListener('pointercancel', release);

  function frame() {
    if (!drag) {
      if (snapping) {
        var d = target - ang;
        ang += d * 0.16;
        if (Math.abs(d) < 0.0004) { ang = target; snapping = false; }
      } else if (vel !== 0) {
        ang += vel; vel *= 0.93;
        if (Math.abs(vel) < 0.006) { vel = 0; target = WL.snapTarget(ang, items.length); snapping = true; }
      }
    }
    var cur = WL.selectedIndex(ang, items.length);
    if (cur !== sel) { sel = cur; if (typeof WISH.onSelect === 'function') WISH.onSelect(cur); }
    for (var i = 0; i < els.length; i++) {
      var l = WL.layout(ang, i, items.length, GEOM);
      els[i].style.transform = 'translate(' + l.x.toFixed(1) + 'px,' + l.y.toFixed(1) +
                               'px) scale(' + l.scale.toFixed(3) + ')';
      els[i].style.opacity = l.opacity.toFixed(2);
      els[i].style.zIndex = l.z;
    }
    requestAnimationFrame(frame);
  }

  function resize() { GEOM.cx = orb.clientWidth / 2; }
  window.addEventListener('resize', resize);

  var WISH = globalThis.WISH;
  WISH.orbit = {
    goTo: goTo,
    selected: function () { return sel; },
    setTaken: function (i, taken) { if (els[i]) els[i].classList.toggle('taken', !!taken); }
  };

  var prev = WISH.onReady;
  WISH.onReady = function () {
    if (typeof prev === 'function') prev();
    if (!items.length) {
      document.getElementById('empty').classList.remove('hidden');
      document.getElementById('card').style.display = 'none';
      return;
    }
    document.getElementById('total').textContent = items.length;
    resize(); build(); frame();
  };
})();
</script>
```

```html
<script>
(function () {
  'use strict';
  var WISH = globalThis.WISH;
  var items = Array.isArray(window.ITENS) ? window.ITENS : [];
  var claimed = {};             // itemId -> hashedName
  var subs = [];
  var db = null, ready = false;

  var claims = {
    available: false,
    subscribe: function (fn) { subs.push(fn); fn(claimed); },
    isMine: function (id) {
      return !!(WISH.identity && claimed[id] && claimed[id] === WISH.identity.key);
    },
    claim: function (id) {
      if (!db) return Promise.reject(new Error('offline'));
      return db.ref('claims/' + id).set(WISH.identity.key);
    },
    unclaim: function (id) {
      if (!db) return Promise.reject(new Error('offline'));
      return db.ref('claims/' + id).remove();
    }
  };
  WISH.claims = claims;

  function fire() { subs.forEach(function (fn) { fn(claimed); }); }

  function connect() {
    if (!window.FIREBASE_CONFIG || !window.firebase) return degrade();
    try {
      firebase.initializeApp(window.FIREBASE_CONFIG);
      db = firebase.database();
    } catch (e) { return degrade(); }

    var timer = setTimeout(function () { if (!ready) degrade(); }, 3000);
    db.ref('claims').on('value', function (snap) {
      ready = true; clearTimeout(timer);
      claims.available = true;
      claimed = snap.val() || {};
      fire();
    }, function () { clearTimeout(timer); degrade(); });
  }

  function degrade() {
    claims.available = false;
    db = null;
    claimed = {};
    var sub = document.querySelector('header .sub');
    if (sub && !document.getElementById('offline')) {
      var s = document.createElement('span');
      s.id = 'offline';
      s.textContent = ' · reservas indisponíveis';
      sub.appendChild(s);
    }
    fire();
  }

  function stateOf(i) {
    var id = items[i] && items[i].id;
    if (!id || !claimed[id]) return 'available';
    return claims.isMine(id) ? 'mine' : 'taken';
  }

  function repaint() {
    var free = 0;
    for (var i = 0; i < items.length; i++) {
      var taken = !!claimed[items[i].id];
      if (!taken) free++;
      WISH.orbit.setTaken(i, taken);
    }
    document.getElementById('left').textContent = free;
    var cur = WISH.card.current();
    if (cur >= 0) WISH.card.render(cur, stateOf(cur));
  }

  WISH.onSelect = function (i) { WISH.card.render(i, stateOf(i)); };

  WISH.card.onAction(function (i, state) {
    var it = items[i];
    if (!it || !claims.available) { WISH.toast('Reservas indisponíveis agora'); return; }
    var id = it.id, undo = claimed[id];

    if (state === 'available') {
      claimed[id] = WISH.identity.key;           // optimistic
      repaint();
      claims.claim(id).catch(function () {
        if (undo === undefined) delete claimed[id]; else claimed[id] = undo;
        repaint();
        WISH.toast('Não deu pra reservar. Alguém pegou antes?');
      });
    } else if (state === 'mine') {
      delete claimed[id];                        // optimistic
      repaint();
      claims.unclaim(id).catch(function () {
        claimed[id] = undo;
        repaint();
        WISH.toast('Não deu pra desmarcar. Tenta de novo?');
      });
    }
  });

  var prev = WISH.onReady;
  WISH.onReady = function () {
    if (typeof prev === 'function') prev();
    if (!items.length) return;
    claims.subscribe(repaint);
    connect();
  };
})();
</script>
```
